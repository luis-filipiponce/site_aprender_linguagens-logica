📜 História & Evolução

Mais de cinco décadas de padronização e evolução constante.

1972

Criação no Bell Labs

Dennis Ritchie desenvolve a linguagem C como sucessora do B, para reescrever o sistema operacional UNIX.

1978

"The C Programming Language"

Brian Kernighan e Dennis Ritchie publicam o livro K&R, que se torna a referência e define o "K&R C".

1989

ANSI C (C89/C90)

Primeira padronização oficial pela ANSI e depois pela ISO, garantindo portabilidade entre compiladores.

1999

C99

Tipos int64 (long long), bool, VLAs, comentários // e declaração de variáveis em qualquer ponto.

2011

C11

Suporte nativo a threads, operações atômicas (_Atomic), Unicode e _Generic.

2018

C17

Versão de correção de bugs (bugfix release) do C11, sem novos recursos de linguagem.

2023

C23

Novidades modernas: atributos [[...]], typeof, #embed, nullptr, constexpr e bool como palavra-chave.

Um ponteiro é uma variável que guarda o endereço de memória de outra variável. O operador & obtém o endereço e * acessa o valor apontado (desreferência).

ponteiros.c
#include <stdio.h>

void trocar(int *a, int *b) {
    int tmp = *a;
    *a = *b;
    *b = tmp;
}

int main(void) {
    int x = 10, y = 20;
    int *p = &x;               // p guarda o endereço de x

    printf("x = %d, endereço = %p\n", *p, (void*)p);
    *p = 99;                    // altera x via ponteiro
    printf("x agora vale %d\n", x);

    trocar(&x, &y);           // passagem por referência
    printf("x = %d, y = %d\n", x, y);
    return 0;
}

Com malloc, calloc e realloc você reserva memória no heap em tempo de execução. Tudo que for alocado deve ser liberado com free, caso contrário ocorre memory leak.

malloc.c
#include <stdio.h>
#include <stdlib.h>

int main(void) {
    size_t n = 5;
    int *v = malloc(n * sizeof(*v));
    if (v == NULL) {
        perror("malloc");
        return 1;
    }

    for (size_t i = 0; i < n; i++) v[i] = (int)(i * i);

    // aumenta o vetor para 10 elementos
    int *tmp = realloc(v, 10 * sizeof(*v));
    if (tmp == NULL) { free(v); return 1; }
    v = tmp;

    printf("v[4] = %d\n", v[4]);
    free(v);        // sempre libere!
    v = NULL;
    return 0;
}

struct agrupa campos de tipos diferentes, cada um com seu próprio espaço. Em uma union, todos os campos compartilham a mesma memória, e o tamanho é o do maior campo.

structs.c
#include <stdio.h>
#include <stdint.h>

typedef struct {
    char   nome[50];
    int    idade;
    float  nota;
} Aluno;

typedef union {
    uint32_t inteiro;
    float    decimal;
    unsigned char bytes[4];
} Valor;

int main(void) {
    Aluno a = { "Maria", 20, 9.5f };
    Aluno *pa = &a;
    printf("%s tem %d anos (nota %.1f)\n", pa->nome, pa->idade, pa->nota);

    Valor v;
    v.decimal = 1.0f;
    printf("bits de 1.0f = 0x%08X\n", v.inteiro);
    printf("sizeof(Aluno)=%zu sizeof(Valor)=%zu\n", sizeof(Aluno), sizeof(Valor));
    return 0;
}

Somar 1 a um ponteiro avança sizeof(tipo) bytes. Por isso arr[i] é equivalente a *(arr + i). Subtrair dois ponteiros do mesmo array resulta no número de elementos entre eles.

aritmetica.c
#include <stdio.h>
#include <stddef.h>

int main(void) {
    int arr[] = {10, 20, 30, 40, 50};
    int *p = arr;                  // aponta para arr[0]
    int *fim = arr + 5;

    printf("*(p+2) = %d\n", *(p + 2));   // 30

    for (; p < fim; p++)
        printf("%d ", *p);
    printf("\n");

    ptrdiff_t dist = fim - arr;        // 5 elementos
    printf("distância = %td\n", dist);
    return 0;
}

Arrays são blocos contíguos de memória. Em C, uma string é um array de char terminado pelo caractere nulo '\0'. Sempre cuide dos limites para evitar buffer overflow.

strings.c
#include <stdio.h>
#include <string.h>
#include <ctype.h>

int main(void) {
    int notas[3][2] = { {7, 8}, {9, 6}, {10, 9} };
    printf("notas[2][1] = %d\n", notas[2][1]);

    char s[32] = "linguagem";
    strncat(s, " C", sizeof(s) - strlen(s) - 1);

    for (size_t i = 0; s[i] != '\0'; i++)
        s[i] = (char)toupper((unsigned char)s[i]);

    printf("%s (%zu caracteres)\n", s, strlen(s));

    if (strcmp(s, "LINGUAGEM C") == 0)
        puts("Strings iguais!");
    return 0;
}

Funções organizam o código em blocos reutilizáveis. Uma função recursiva chama a si mesma e precisa de um caso base para terminar. Cada chamada cria um novo quadro (stack frame) na pilha.

recursao.c
#include <stdio.h>

unsigned long long fatorial(unsigned int n) {
    if (n <= 1) return 1;            // caso base
    return n * fatorial(n - 1);       // chamada recursiva
}

int fibonacci(int n) {
    return (n < 2) ? n : fibonacci(n - 1) + fibonacci(n - 2);
}

int mdc(int a, int b) {            // algoritmo de Euclides
    return b == 0 ? a : mdc(b, a % b);
}

int main(void) {
    printf("10! = %llu\n", fatorial(10));
    printf("fib(10) = %d\n", fibonacci(10));
    printf("mdc(48,18) = %d\n", mdc(48, 18));
    return 0;
}

Lista encadeada simples com inserção no início e no fim, busca, remoção e liberação de memória.

lista.c
#include <stdio.h>
#include <stdlib.h>

typedef struct No {
    int valor;
    struct No *proximo;
} No;

typedef struct {
    No *inicio;
    size_t tamanho;
} Lista;

void lista_init(Lista *l) {
    l->inicio = NULL;
    l->tamanho = 0;
}

static No *novo_no(int valor) {
    No *n = malloc(sizeof(*n));
    if (!n) { perror("malloc"); exit(EXIT_FAILURE); }
    n->valor = valor;
    n->proximo = NULL;
    return n;
}

void lista_push_front(Lista *l, int valor) {
    No *n = novo_no(valor);
    n->proximo = l->inicio;
    l->inicio = n;
    l->tamanho++;
}

void lista_push_back(Lista *l, int valor) {
    No *n = novo_no(valor);
    if (!l->inicio) {
        l->inicio = n;
    } else {
        No *atual = l->inicio;
        while (atual->proximo) atual = atual->proximo;
        atual->proximo = n;
    }
    l->tamanho++;
}

No *lista_buscar(const Lista *l, int valor) {
    for (No *p = l->inicio; p; p = p->proximo)
        if (p->valor == valor) return p;
    return NULL;
}

int lista_remover(Lista *l, int valor) {
    No *anterior = NULL, *atual = l->inicio;
    while (atual && atual->valor != valor) {
        anterior = atual;
        atual = atual->proximo;
    }
    if (!atual) return 0;                 // não encontrado
    if (anterior) anterior->proximo = atual->proximo;
    else          l->inicio = atual->proximo;
    free(atual);
    l->tamanho--;
    return 1;
}

void lista_imprimir(const Lista *l) {
    for (No *p = l->inicio; p; p = p->proximo)
        printf("%d -> ", p->valor);
    printf("NULL (%zu elementos)\n", l->tamanho);
}

void lista_destruir(Lista *l) {
    No *p = l->inicio;
    while (p) {
        No *prox = p->proximo;
        free(p);
        p = prox;
    }
    l->inicio = NULL;
    l->tamanho = 0;
}

int main(void) {
    Lista l;
    lista_init(&l);

    lista_push_back(&l, 10);
    lista_push_back(&l, 20);
    lista_push_back(&l, 30);
    lista_push_front(&l, 5);
    lista_imprimir(&l);                 // 5 -> 10 -> 20 -> 30 -> NULL

    printf("Busca 20: %s\n", lista_buscar(&l, 20) ? "achou" : "não achou");

    lista_remover(&l, 10);
    lista_imprimir(&l);                 // 5 -> 20 -> 30 -> NULL

    lista_destruir(&l);
    return 0;
}

Gravação e leitura de registros em um arquivo binário usando fwrite, fread e fseek.

arquivo_binario.c
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define ARQUIVO "alunos.dat"

typedef struct {
    int   id;
    char  nome[40];
    float nota;
} Aluno;

int salvar_alunos(const char *caminho, const Aluno *v, size_t n) {
    FILE *f = fopen(caminho, "wb");
    if (!f) { perror("fopen"); return -1; }

    size_t escritos = fwrite(v, sizeof(Aluno), n, f);
    fclose(f);
    return escritos == n ? 0 : -1;
}

long contar_registros(FILE *f) {
    fseek(f, 0, SEEK_END);
    long bytes = ftell(f);
    rewind(f);
    return bytes / (long)sizeof(Aluno);
}

int main(void) {
    Aluno turma[] = {
        { 1, "Ana Souza",    9.0f },
        { 2, "Bruno Lima",   7.5f },
        { 3, "Carla Mendes", 8.8f }
    };
    size_t total = sizeof(turma) / sizeof(turma[0]);

    if (salvar_alunos(ARQUIVO, turma, total) != 0) {
        fputs("Erro ao gravar arquivo\n", stderr);
        return EXIT_FAILURE;
    }
    printf("%zu registros gravados em %s\n", total, ARQUIVO);

    FILE *f = fopen(ARQUIVO, "rb");
    if (!f) { perror("fopen"); return EXIT_FAILURE; }

    printf("Arquivo contém %ld registros\n", contar_registros(f));

    Aluno a;
    while (fread(&a, sizeof(Aluno), 1, f) == 1)
        printf("#%d  %-15s  nota %.1f\n", a.id, a.nome, a.nota);

    // acesso aleatório: lê direto o 2º registro
    fseek(f, (long)sizeof(Aluno) * 1, SEEK_SET);
    if (fread(&a, sizeof(Aluno), 1, f) == 1)
        printf("Registro 2: %s\n", a.nome);

    fclose(f);
    return EXIT_SUCCESS;
}

Pilha (LIFO) com array dinâmico que cresce automaticamente via realloc, usada para verificar parênteses balanceados.

pilha.c
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>

typedef struct {
    int   *dados;
    size_t topo;        // quantidade de elementos
    size_t capacidade;
} Pilha;

Pilha *pilha_criar(size_t capacidade_inicial) {
    Pilha *p = malloc(sizeof(*p));
    if (!p) return NULL;
    p->dados = malloc(capacidade_inicial * sizeof(int));
    if (!p->dados) { free(p); return NULL; }
    p->topo = 0;
    p->capacidade = capacidade_inicial;
    return p;
}

bool pilha_vazia(const Pilha *p) { return p->topo == 0; }

bool pilha_push(Pilha *p, int valor) {
    if (p->topo == p->capacidade) {
        size_t nova = p->capacidade * 2;
        int *tmp = realloc(p->dados, nova * sizeof(int));
        if (!tmp) return false;
        p->dados = tmp;
        p->capacidade = nova;
    }
    p->dados[p->topo++] = valor;
    return true;
}

bool pilha_pop(Pilha *p, int *saida) {
    if (pilha_vazia(p)) return false;
    *saida = p->dados[--p->topo];
    return true;
}

bool pilha_peek(const Pilha *p, int *saida) {
    if (pilha_vazia(p)) return false;
    *saida = p->dados[p->topo - 1];
    return true;
}

void pilha_destruir(Pilha *p) {
    if (!p) return;
    free(p->dados);
    free(p);
}

// Verifica se (), [] e {} estão balanceados
bool balanceado(const char *expr) {
    Pilha *p = pilha_criar(4);
    if (!p) return false;
    bool ok = true;

    for (const char *c = expr; *c && ok; c++) {
        if (*c == '(' || *c == '[' || *c == '{') {
            ok = pilha_push(p, *c);
        } else if (*c == ')' || *c == ']' || *c == '}') {
            int abre;
            if (!pilha_pop(p, &abre)) { ok = false; break; }
            ok = (abre == '(' && *c == ')') ||
                 (abre == '[' && *c == ']') ||
                 (abre == '{' && *c == '}');
        }
    }
    ok = ok && pilha_vazia(p);
    pilha_destruir(p);
    return ok;
}

int main(void) {
    Pilha *p = pilha_criar(2);
    for (int i = 1; i <= 5; i++) pilha_push(p, i * 10);

    int v;
    while (pilha_pop(p, &v)) printf("pop: %d\n", v);
    pilha_destruir(p);

    printf("\"{[()]}\" balanceado? %s\n", balanceado("{[()]}") ? "sim" : "não");
    printf("\"([)]\"   balanceado? %s\n", balanceado("([)]")   ? "sim" : "não");
    return 0;
}
