JavaScript é single-thread: executa uma coisa por vez na Call Stack. Tarefas assíncronas (timers, rede) são delegadas ao ambiente e voltam por filas. O Event Loop move callbacks para a stack quando ela está vazia — microtasks (Promises) têm prioridade sobre macrotasks (setTimeout).

event-loop.js
console.log('1. início');

setTimeout(() => console.log('4. macrotask (setTimeout)'), 0);

Promise.resolve().then(() => console.log('3. microtask (Promise)'));

console.log('2. fim do código síncrono');

// Saída: 1 → 2 → 3 → 4

Uma closure é uma função que "lembra" das variáveis do escopo onde foi criada, mesmo depois que esse escopo terminou. É a base de encapsulamento, fábricas de funções e hooks.

closures.js
function criarContador(inicial = 0) {
  let contagem = inicial;       // variável "privada"

  return {
    incrementar: () => ++contagem,
    decrementar: () => --contagem,
    valor:       () => contagem,
  };
}

const contador = criarContador(10);
contador.incrementar();
contador.incrementar();
console.log(contador.valor()); // 12
console.log(contador.contagem); // undefined — inacessível de fora

Uma Promise representa um valor que estará disponível no futuro (pending → fulfilled/rejected). async/await é açúcar sintático que permite escrever fluxos assíncronos com aparência de código síncrono.

async.js
const esperar = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

async function processar() {
  try {
    console.log('Iniciando...');
    await esperar(1000);

    // Executa em paralelo
    const [a, b] = await Promise.all([
      fetch('/api/a').then((r) => r.json()),
      fetch('/api/b').then((r) => r.json()),
    ]);
    return { a, b };
  } catch (erro) {
    console.error('Falhou:', erro.message);
  } finally {
    console.log('Finalizado');
  }
}

Em JavaScript, objetos herdam de outros objetos através do protótipo. Ao acessar uma propriedade, o motor procura no objeto e sobe pela cadeia (__proto__) até encontrá-la ou chegar a null. A sintaxe class é açúcar sobre esse mecanismo.

prototype.js
class Animal {
  constructor(nome) { this.nome = nome; }
  falar() { return `${this.nome} faz um som`; }
}

class Cachorro extends Animal {
  falar() { return `${this.nome} late: Au au!`; }
}

const rex = new Cachorro('Rex');
console.log(rex.falar());                                   // Rex late: Au au!
console.log(Object.getPrototypeOf(rex) === Cachorro.prototype); // true
console.log(rex instanceof Animal);                           // true
console.log(rex.hasOwnProperty('falar'));                   // false — vem do protótipo

Destructuring extrai valores de objetos e arrays em variáveis de forma concisa. Spread (...) expande iteráveis; com rest, agrupa o restante. Essenciais para código imutável.

destructuring.js
const usuario = { nome: 'Ana', idade: 28, endereco: { cidade: 'São Paulo' } };

// Destructuring com renomeação, valor padrão e aninhamento
const { nome, idade: anos, pais = 'Brasil', endereco: { cidade } } = usuario;

// Arrays
const [primeiro, , terceiro, ...resto] = [1, 2, 3, 4, 5];
console.log(primeiro, terceiro, resto); // 1 3 [4, 5]

// Spread: cópia imutável com alteração
const atualizado = { ...usuario, idade: 29 };
const lista = [0, ...resto, 99];

// Parâmetros com destructuring
const saudacao = ({ nome, pais = 'Brasil' }) => `Olá, ${nome} do ${pais}!`;

O DOM é a representação do HTML como uma árvore de objetos (nós) que o JavaScript pode ler e modificar. Tudo começa em document. Selecione com querySelector/querySelectorAll, crie nós com createElement e altere classes com classList. Segurança: prefira textContent a innerHTML ao exibir dados de usuários — innerHTML abre brecha para ataques XSS.

dom.js
// 1. Seleção de elementos
const titulo = document.querySelector('#titulo');          // primeiro que casar (ou null)
const itens  = document.querySelectorAll('.lista li');    // NodeList estática

// 2. Leitura e alteração de conteúdo, atributos e estilo
titulo.textContent = 'Olá, DOM!';                 // seguro: trata como texto puro
titulo.setAttribute('data-status', 'ativo');
titulo.classList.add('destaque');                 // também: remove, toggle, contains
titulo.style.color = 'tomato';
console.log(titulo.dataset.status);               // "ativo" (atributos data-*)

// 3. Criação e inserção de nós
const lista = document.querySelector('.lista');
const novo  = document.createElement('li');
novo.textContent = 'Novo item';
lista.append(novo);                                  // ao final
lista.prepend('Primeiro');                         // ao início (aceita texto)
novo.after(document.createElement('li'));          // irmão seguinte

// 4. Navegação pela árvore
console.log(lista.parentElement, lista.children, lista.firstElementChild);
console.log(novo.closest('.lista'));                  // ancestral mais próximo que casa
novo.remove();                                       // remove do DOM

// 5. Várias inserções de uma vez (evita reflow repetido)
const frag = document.createDocumentFragment();
['A', 'B', 'C'].forEach((t) => {
  const li = document.createElement('li');
  li.textContent = t;
  frag.append(li);
});
lista.append(frag);

Eventos conectam o usuário ao código (click, input, submit, keydown…). Um evento percorre captura → alvo → bubbling (sobe pelos ancestrais). Isso permite a delegação de eventos: um único listener no pai atende todos os filhos, inclusive os criados depois. Use preventDefault() para cancelar o comportamento padrão e stopPropagation() com parcimônia.

eventos.js
const form  = document.querySelector('#form');
const lista = document.querySelector('#tarefas');

// Formulário: cancela o envio padrão e lê os dados
form.addEventListener('submit', (e) => {
  e.preventDefault();
  const dados = Object.fromEntries(new FormData(form));
  console.log(dados);                       // { titulo: '...' }
});

// Delegação: UM listener para todos os botões "remover", presentes e futuros
lista.addEventListener('click', (e) => {
  const botao = e.target.closest('button.remover');
  if (!botao || !lista.contains(botao)) return;
  botao.closest('li').remove();
});

// Opções do listener: once (roda uma vez) e AbortController (remoção em lote)
const ctrl = new AbortController();
window.addEventListener('resize', () => console.log(innerWidth), { signal: ctrl.signal });
document.addEventListener('keydown', (e) => {
  if (e.key === 'Escape') ctrl.abort();    // remove os listeners ligados ao signal
});

// Eventos personalizados
lista.addEventListener('tarefa:criada', (e) => console.log(e.detail));
lista.dispatchEvent(new CustomEvent('tarefa:criada', { detail: { id: 1 } }));

// Rodar só quando o DOM estiver pronto (se o script não for type="module"/defer)
document.addEventListener('DOMContentLoaded', () => console.log('DOM pronto'));

localStorage persiste dados no navegador sem expirar; sessionStorage vale apenas para a aba aberta. Ambos guardam somente strings, por isso usamos JSON.stringify/JSON.parse. Segurança: nunca guarde senhas, tokens sensíveis ou dados pessoais no Web Storage — qualquer script da página consegue lê-lo.

storage.js
const CHAVE = 'preferencias';

function salvar(prefs) {
  try {
    localStorage.setItem(CHAVE, JSON.stringify(prefs));
  } catch (erro) {
    console.warn('Armazenamento indisponível ou cheio', erro.name); // QuotaExceededError
  }
}

function carregar() {
  try {
    return JSON.parse(localStorage.getItem(CHAVE)) ?? { tema: 'claro' };
  } catch {
    return { tema: 'claro' };                  // JSON corrompido → valor padrão
  }
}

salvar({ tema: 'escuro', fonte: 16 });
console.log(carregar().tema);                  // "escuro"

localStorage.removeItem(CHAVE);                    // remove uma chave

// Reage a mudanças feitas em OUTRAS abas
window.addEventListener('storage', (e) => console.log(e.key, e.newValue));

O navegador oferece APIs além do DOM básico. IntersectionObserver detecta quando um elemento entra na tela (lazy loading, animações ao rolar) sem escutar scroll; MutationObserver observa mudanças no DOM; ResizeObserver reage a mudanças de tamanho. Também são úteis URL/URLSearchParams, History API e requestAnimationFrame para animações.

web-apis.js
// Lazy loading de imagens com IntersectionObserver
const observador = new IntersectionObserver((entradas, obs) => {
  entradas.forEach((entrada) => {
    if (!entrada.isIntersecting) return;
    const img = entrada.target;
    img.src = img.dataset.src;          // carrega só quando visível
    obs.unobserve(img);
  });
}, { rootMargin: '200px' });

document.querySelectorAll('img[data-src]').forEach((img) => observador.observe(img));

// URL e parâmetros de consulta, sem concatenar strings
const url = new URL('https://exemplo.com/busca');
url.searchParams.set('q', 'javascript dom');
console.log(url.href);                         // https://exemplo.com/busca?q=javascript+dom

// Animação sincronizada com a taxa de atualização da tela
function animar(tempo) {
  // ...atualizar posição com base em "tempo"...
  requestAnimationFrame(animar);
}
requestAnimationFrame(animar);

// Debounce: evita disparar a cada tecla (busca, resize)
const debounce = (fn, ms = 300) => {
  let t;
  return (...args) => { clearTimeout(t); t = setTimeout(() => fn(...args), ms); };
};
document.querySelector('#busca').addEventListener('input', debounce((e) => console.log(e.target.value)));

Os métodos de ordem superior transformam coleções sem laços manuais nem mutação: map (transforma), filter (seleciona), reduce (acumula), find/some/every (busca e testes). Combine-os em cadeia (pipeline). Atenção: sort(), push() e splice() mutam o array original — prefira toSorted(), toSpliced() ou uma cópia.

arrays.js
const produtos = [
  { nome: 'Notebook', preco: 3500, estoque: 2 },
  { nome: 'Mouse',    preco: 80,   estoque: 0 },
  { nome: 'Teclado',  preco: 250,  estoque: 5 },
];

const total = produtos
  .filter((p) => p.estoque > 0)                  // só disponíveis
  .map((p) => p.preco * 0.85)                   // 15% de desconto
  .reduce((soma, preco) => soma + preco, 0);    // total
console.log(total.toFixed(2));                    // "3187.50"

console.log(produtos.find((p) => p.nome === 'Mouse'));
console.log(produtos.some((p) => p.estoque === 0));    // true
console.log(produtos.every((p) => p.preco > 50));    // true

// Ordenar uma CÓPIA, sem mutar o original
const porPreco = [...produtos].sort((a, b) => a.preco - b.preco);
const nomes    = produtos.map(({ nome }) => nome);
console.log(porPreco[0].nome, nomes);              // Mouse [ 'Notebook', 'Mouse', 'Teclado' ]

JavaScript tem tipagem dinâmica e coerção implícita. Os tipos primitivos são string, number, bigint, boolean, undefined, null e symbol; o resto é objeto. Use sempre ===/!== (sem coerção) e conheça os valores falsy: false, 0, '', null, undefined, NaN. Para números decimais, lembre que 0.1 + 0.2 !== 0.3 (ponto flutuante IEEE 754).

tipos.js
console.log(typeof 'a', typeof 1, typeof undefined, typeof null); // string number undefined object (!)
console.log(0 == '', 0 === '');            // true false — prefira ===
console.log('5' + 1, '5' - 1);              // "51" 4 — + concatena, - converte
console.log(Number('abc'), Number.isNaN(NaN)); // NaN true
console.log(0.1 + 0.2);                    // 0.30000000000000004
console.log(Math.abs(0.1 + 0.2 - 0.3) < Number.EPSILON); // true — compare com tolerância

// || trata "falsy" como ausente; ?? só trata null/undefined
const quantidade = 0;
console.log(quantidade || 10, quantidade ?? 10);   // 10 0

// Objetos e arrays são copiados por REFERÊNCIA
const a = [1, 2];
const b = a;                  // mesma referência
const c = structuredClone(a);   // cópia profunda independente
b.push(3);
console.log(a, c);               // [1, 2, 3] [1, 2]

O valor de this depende de como a função é chamada, não de onde foi escrita. Em métodos, aponta para o objeto; em chamadas soltas, é undefined (strict mode). Arrow functions não têm this próprio — herdam o do escopo externo. Para fixar o contexto, use bind, call ou apply.

this.js
'use strict'; // sem strict mode, "this" solto seria window em vez de undefined

const pessoa = {
  nome: 'Ana',
  falar() { return `Oi, sou ${this.nome}`; },          // this = pessoa
  falarArrow: () => typeof this,                       // this do escopo externo, NÃO pessoa
  falarDepois() {
    setTimeout(() => console.log(this.nome), 100); // arrow preserva this → "Ana"
  },
};

console.log(pessoa.falar());             // "Oi, sou Ana"

const solta = pessoa.falar;              // perdeu o contexto!
try { solta(); } catch (e) { console.log(e.name); } // TypeError (this é undefined)

const presa = pessoa.falar.bind(pessoa);  // this fixo
console.log(presa());                      // "Oi, sou Ana"
console.log(pessoa.falar.call({ nome: 'Beto' })); // "Oi, sou Beto"

// Em listeners do DOM: função normal → this = elemento; arrow → use e.currentTarget
document.querySelector('button').addEventListener('click', function () {
  console.log(this === document.querySelector('button')); // true
});

Os operadores relacionais (>, <, >=, <=, ===, !==) comparam valores. Os lógicos (&&, ||, !, ??) operam com curto-circuito (short-circuit evaluation). O operador ternário (condição ? valorSeVerdadeiro : valorSeFalso) é uma expressão condicional compacta ideal para retornos rápidos e renderização condicional.

operadores.js
// 1. Operadores Relacionais (sempre use === e !==)
const idade = 20;
const maiorDeIdade = idade >= 18; // true
const ehIgual = ("10" === 10);     // false (tipos diferentes: string vs number)

// 2. Operadores Lógicos e Curto-Circuito
const temCarteira = true;
const podeDirigir = maiorDeIdade && temCarteira; // true (E lógico)
const temAcesso = podeDirigir || false;          // true (OU lógico)
const bloqueado = !temAcesso;                   // false (NÃO lógico)

// Curto-circuito avalia apenas o necessário
const usuario = { nome: "Carlos" };
usuario && console.log(`Usuário logado: ${usuario.nome}`);

// 3. Operador Ternário (condicao ? verdadeiro : falso)
const status = idade >= 18 ? "Adulto" : "Menor de idade";
console.log(status); // "Adulto"

// Ternários aninhados (use com moderação)
const nota = 85;
const conceito = nota >= 90 ? "A" : nota >= 70 ? "B" : "C";
console.log(conceito); // "B"

Com if, else if e else você bifurca o fluxo de execução baseado em expressões booleanas. A estrutura switch / case compara valores com igualdade estrita (===) e é excelente para múltiplos estados discretos, inclusive permitindo avaliar expressões com switch (true).

condicoes.js
// 1. if / else if / else
const hora = 14;
if (hora < 12) {
  console.log("Bom dia!");
} else if (hora < 18) {
  console.log("Boa tarde!");
} else {
  console.log("Boa noite!");
}

// 2. switch / case tradicional
const perfil = "admin";
switch (perfil) {
  case "admin":
    console.log("Acesso total concedido");
    break;
  case "editor":
    console.log("Permissão para criar e editar");
    break;
  case "leitor":
    console.log("Permissão apenas de leitura");
    break;
  default:
    console.log("Perfil desconhecido");
}

// 3. Padrão switch (true) para faixas de valores
const pontuacao = 82;
switch (true) {
  case pontuacao >= 90:
    console.log("Excelente");
    break;
  case pontuacao >= 70:
    console.log("Aprovado");
    break;
  default:
    console.log("Em recuperação");
}

Laços iteram sobre blocos de código: for tradicional com contador, while (enquanto condição for verdadeira), do...while (executa ao menos uma vez antes do teste), for...of (percorre valores de iteráveis como Arrays, Maps e Sets) e for...in (percorre chaves enumeráveis de objetos). Use break para interromper e continue para pular para a próxima iteração.

lacos.js
// 1. for tradicional
for (let i = 1; i <= 3; i++) {
  console.log(`Contador: ${i}`);
}

// 2. while e do...while
let vidas = 3;
while (vidas > 0) {
  console.log(`Restam ${vidas} vidas`);
  vidas--;
}

let tentativa = 0;
do {
  tentativa++;
  console.log(`Tentativa executada: ${tentativa}`);
} while (tentativa < 1);

// 3. for...of (iteráveis: Array, Set, Map, String)
const linguagens = ["JavaScript", "Python", "Java"];
for (const lang of linguagens) {
  if (lang === "Python") continue; // pula iteração
  console.log(`Linguagem: ${lang}`);
}

// 4. for...in (chaves/propriedades de objetos)
const config = { tema: "escuro", zoom: 100, debug: true };
for (const chave in config) {
  console.log(`${chave}: ${config[chave]}`);
}

O bloco try...catch...finally impede que exceções não tratadas travem a aplicação. Com throw você dispara exceções (nativas ou classes herdadas de Error), e o bloco finally sempre roda, garantindo liberação de recursos.

excecoes.js
// 1. Exceção customizada
class ValidacaoError extends Error {
  constructor(mensagem, campo) {
    super(mensagem);
    this.name = "ValidacaoError";
    this.campo = campo;
  }
}

function cadastrarUsuario(dados) {
  if (!dados.email || !dados.email.includes("@")) {
    throw new ValidacaoError("E-mail inválido", "email");
  }
  return { id: 1, ...dados };
}

// 2. Captura e tratamento estruturado
try {
  const novoUser = cadastrarUsuario({ nome: "Ana", email: "invalido" });
  console.log(novoUser);
} catch (erro) {
  if (erro instanceof ValidacaoError) {
    console.error(`Erro no campo [${erro.campo}]: ${erro.message}`);
  } else {
    console.error("Erro inesperado:", erro.message);
  }
} finally {
  console.log("Operação concluída (limpeza de recursos).");
}

Em JavaScript, primitivos (string, number, boolean, symbol, bigint) não são objetos. Porém, ao chamar métodos como "texto".toUpperCase() ou (12.34).toFixed(1), a engine JS realiza temporariamente um auto-boxing, envolvendo o valor no objeto wrapper correspondente (new String(), new Number()) e descartando-o logo em seguida. Use as funções construtoras como conversores (ex: String(123), Number("42")) e nunca com new.

wrappers.js
// 1. Primitivo vs Wrapper Object
const primStr = "javascript";
const objStr  = new String("javascript"); // Evite new String()!

console.log(typeof primStr); // "string"
console.log(typeof objStr);  // "object"
console.log(primStr === objStr); // false

// 2. Auto-boxing em ação
// Ao acessar propriedades, o JS cria o wrapper temporário:
console.log(primStr.length);          // 10
console.log(primStr.toUpperCase());   // "JAVASCRIPT"

// 3. Unboxing explícito com .valueOf()
const primitivo = objStr.valueOf();
console.log(typeof primitivo); // "string"

// 4. Uso correto de Wrappers como funções de conversão
const num = Number("123");      // 123 (number primitivo)
const str = String(true);       // "true" (string primitivo)
const bool = Boolean(1);        // true (boolean primitivo)

O JavaScript oferece suporte a Classes com construtores (constructor), métodos de instância, métodos estáticos (static), getters/setters e campos privados reais com o prefixo #. A abstração é alcançada por contratos de classes base que impedem instanciação direta ou forçam subclasses a implementar métodos específicos.

classes_abstracao.js
// Classe base abstrata
class FormaGeometrica {
  constructor(nome) {
    if (new.target === FormaGeometrica) {
      throw new Error("Classe abstrata não pode ser instanciada diretamente!");
    }
    this.nome = nome;
  }

  // Método abstrato que DEVE ser implementado pelas subclasses
  calcularArea() {
    throw new Error("Método abstrato calcularArea() deve ser implementado!");
  }

  // Método de instância compartilhado
  descrever() {
    return `${this.nome} com área de ${this.calcularArea().toFixed(2)} cm²`;
  }
}

// Subclasse concreta com encapsulamento privado (#)
class Circulo extends FormaGeometrica {
  #raio; // Campo estritamente privado

  constructor(raio) {
    super("Círculo");
    this.#raio = raio;
  }

  // Implementação do método abstrato
  calcularArea() {
    return Math.PI * (this.#raio ** 2);
  }

  // Método estático utilitário
  static criarComDiametro(diametro) {
    return new Circulo(diametro / 2);
  }
}

Consumo de API com fetch, tratamento de erros HTTP, timeout via AbortController e retentativas.

api-client.js
class ApiError extends Error {
  constructor(mensagem, status) {
    super(mensagem);
    this.name = 'ApiError';
    this.status = status;
  }
}

async function requisitar(url, { timeout = 8000, tentativas = 3, ...opcoes } = {}) {
  for (let i = 1; i <= tentativas; i++) {
    const controller = new AbortController();
    const timer = setTimeout(() => controller.abort(), timeout);

    try {
      const resposta = await fetch(url, { ...opcoes, signal: controller.signal });

      if (!resposta.ok) {
        throw new ApiError(`HTTP ${resposta.status}: ${resposta.statusText}`, resposta.status);
      }
      return await resposta.json();
    } catch (erro) {
      const ehTimeout = erro.name === 'AbortError';
      const ehClient = erro instanceof ApiError && erro.status < 500;

      // Erros 4xx não adiantam repetir; esgotou tentativas → propaga
      if (ehClient || i === tentativas) {
        throw ehTimeout ? new ApiError('Tempo esgotado', 408) : erro;
      }
      console.warn(`Tentativa ${i} falhou, tentando novamente...`);
      await new Promise((r) => setTimeout(r, 500 * 2 ** i)); // backoff exponencial
    } finally {
      clearTimeout(timer);
    }
  }
}

// Uso
async function carregarUsuarios() {
  try {
    const usuarios = await requisitar('https://jsonplaceholder.typicode.com/users');
    usuarios.forEach(({ name, email }) => console.log(`${name} — ${email}`));
  } catch (erro) {
    console.error(`[${erro.name}]`, erro.message);
  }
}

Lista de tarefas com useState, useEffect, useMemo e useCallback, persistindo no localStorage.

TodoList.jsx
import { useState, useEffect, useMemo, useCallback } from 'react';

export default function TodoList() {
  const [tarefas, setTarefas] = useState(() =>
    JSON.parse(localStorage.getItem('tarefas') ?? '[]')
  );
  const [texto, setTexto]   = useState('');
  const [filtro, setFiltro] = useState('todas'); // todas | pendentes | feitas

  // Efeito colateral: sincroniza com localStorage
  useEffect(() => {
    localStorage.setItem('tarefas', JSON.stringify(tarefas));
  }, [tarefas]);

  const adicionar = useCallback((e) => {
    e.preventDefault();
    const titulo = texto.trim();
    if (!titulo) return;
    setTarefas((atual) => [...atual, { id: crypto.randomUUID(), titulo, feita: false }]);
    setTexto('');
  }, [texto]);

  const alternar = (id) =>
    setTarefas((atual) => atual.map((t) => (t.id === id ? { ...t, feita: !t.feita } : t)));

  const remover = (id) =>
    setTarefas((atual) => atual.filter((t) => t.id !== id));

  const visiveis = useMemo(() => {
    if (filtro === 'pendentes') return tarefas.filter((t) => !t.feita);
    if (filtro === 'feitas')    return tarefas.filter((t) => t.feita);
    return tarefas;
  }, [tarefas, filtro]);

  return (
    <section className="todo">
      <h2>Minhas Tarefas ({tarefas.filter((t) => !t.feita).length} pendentes)</h2>

      <form onSubmit={adicionar}>
        <input
          value={texto}
          onChange={(e) => setTexto(e.target.value)}
          placeholder="Nova tarefa..."
        />
        <button type="submit">Adicionar</button>
      </form>

      <nav>
        {['todas', 'pendentes', 'feitas'].map((f) => (
          <button key={f} disabled={filtro === f} onClick={() => setFiltro(f)}>
            {f}
          </button>
        ))}
      </nav>

      <ul>
        {visiveis.map(({ id, titulo, feita }) => (
          <li key={id} style={{ textDecoration: feita ? 'line-through' : 'none' }}>
            <input type="checkbox" checked={feita} onChange={() => alternar(id)} />
            {titulo}
            <button onClick={() => remover(id)} aria-label="Remover">✕</button>
          </li>
        ))}
      </ul>
    </section>
  );
}

CRUD completo de livros com Express: validação, middleware de erros e códigos HTTP corretos. Dados em memória para fins didáticos.

server.js
// npm init -y && npm i express
import express from 'express';

const app = express();
const PORTA = process.env.PORT ?? 3000;

app.use(express.json());

// Logger simples
app.use((req, _res, next) => {
  console.log(`${new Date().toISOString()} ${req.method} ${req.url}`);
  next();
});

// "Banco de dados" em memória
let livros = [
  { id: 1, titulo: 'Eloquent JavaScript', autor: 'Marijn Haverbeke' },
  { id: 2, titulo: 'You Don\'t Know JS',   autor: 'Kyle Simpson' },
];
let proximoId = 3;

const validar = (req, res, next) => {
  const { titulo, autor } = req.body ?? {};
  if (!titulo || !autor) {
    return res.status(400).json({ erro: 'Campos "titulo" e "autor" são obrigatórios' });
  }
  next();
};

// READ — lista com busca opcional (?q=)
app.get('/api/livros', (req, res) => {
  const q = req.query.q?.toLowerCase();
  res.json(q ? livros.filter((l) => l.titulo.toLowerCase().includes(q)) : livros);
});

// READ — por id
app.get('/api/livros/:id', (req, res) => {
  const livro = livros.find((l) => l.id === Number(req.params.id));
  if (!livro) return res.status(404).json({ erro: 'Livro não encontrado' });
  res.json(livro);
});

// CREATE
app.post('/api/livros', validar, (req, res) => {
  const { titulo, autor } = req.body;
  const novo = { id: proximoId++, titulo, autor };
  livros.push(novo);
  res.status(201).location(`/api/livros/${novo.id}`).json(novo);
});

// UPDATE
app.put('/api/livros/:id', validar, (req, res) => {
  const indice = livros.findIndex((l) => l.id === Number(req.params.id));
  if (indice === -1) return res.status(404).json({ erro: 'Livro não encontrado' });
  livros[indice] = { ...livros[indice], ...req.body, id: livros[indice].id };
  res.json(livros[indice]);
});

// DELETE
app.delete('/api/livros/:id', (req, res) => {
  const tamanho = livros.length;
  livros = livros.filter((l) => l.id !== Number(req.params.id));
  if (livros.length === tamanho) return res.status(404).json({ erro: 'Livro não encontrado' });
  res.status(204).end();
});

// 404 e tratamento global de erros
app.use((_req, res) => res.status(404).json({ erro: 'Rota não encontrada' }));
app.use((err, _req, res, _next) => {
  console.error(err);
  res.status(500).json({ erro: 'Erro interno do servidor' });
});

app.listen(PORTA, () => console.log(`API rodando em http://localhost:${PORTA}`));
